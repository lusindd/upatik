var FOLDER_ID = '1QihV3D75u0YMhbBE1szWrTp4eEJozRTT'; var SHEET_NAME = 'Sheet1'; var ADMIN_PIN = '1234'; // Header Sheet: // ID | Tanggal | Jurusan | PJ | HP | Jenis | Jumlah | Inventaris | Keluhan | Keterangan | Teknisi | Status | Catatan | TglSelesai | FileURL | FileName | FotoURL | FotoName function doGet(e) { var action = e.parameter.action; var callback = e.parameter.callback; var result; if (action === 'getAll') { result = getAllRecords(); } else if (action === 'getOne') { result = getOneRecord(e.parameter.id); } else if (action === 'checkPin') { result = checkPin(e.parameter.pin); } else if (action === 'submit') { result = submitPermohonan(e.parameter); } else if (action === 'update') { result = updateRecord(e.parameter); } else if (action === 'delete') { result = deleteRecord(e.parameter.id); } else { result = { success: false, message: 'Action tidak dikenal' }; } var json = JSON.stringify(result); var output; if (callback && callback !== 'none') { output = ContentService.createTextOutput(callback + '(' + json + ')'); output.setMimeType(ContentService.MimeType.JAVASCRIPT); } else { output = ContentService.createTextOutput(json); output.setMimeType(ContentService.MimeType.JSON); } return output; } function doPost(e) { var data = JSON.parse(e.postData.contents); var result; if (data.action === 'submit') { result = submitPermohonan(data); } else if (data.action === 'update') { result = updateRecord(data); } else if (data.action === 'delete') { result = deleteRecord(data.id); } else { result = { success: false, message: 'Action tidak dikenal' }; } var output = ContentService.createTextOutput(JSON.stringify(result)); output.setMimeType(ContentService.MimeType.JSON); return output; } function getAllRecords() { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); var data = sheet.getDataRange().getValues(); if (data.length <= 1) return { success: true, data: [] }; var headers = data[0]; var records = []; for (var i = 1; i < data.length; i++) { var row = {}; for (var j = 0; j < headers.length; j++) { row[headers[j]] = data[i][j]; } records.push(row); } return { success: true, data: records }; } catch (err) { return { success: false, message: err.toString() }; } } function getOneRecord(id) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); var data = sheet.getDataRange().getValues(); var headers = data[0]; for (var i = 1; i < data.length; i++) { if (data[i][0] == id) { var row = {}; for (var j = 0; j < headers.length; j++) { row[headers[j]] = data[i][j]; } return { success: true, data: row }; } } return { success: false, message: 'Tiket tidak ditemukan' }; } catch (err) { return { success: false, message: err.toString() }; } } function checkPin(pin) { return { success: pin === ADMIN_PIN }; } function submitPermohonan(data) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); var id = generateId(); var fileUrl = ''; var fileName = ''; if (data.fileBase64 && data.fileName) { var folder = DriveApp.getFolderById(FOLDER_ID); var blob = Utilities.newBlob( Utilities.base64Decode(data.fileBase64), data.fileMime || 'application/pdf', id + '_' + data.fileName ); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); fileUrl = file.getUrl(); fileName = data.fileName; } sheet.appendRow([ id, data.tanggal, data.jurusan, data.pj, data.hp || '', data.jenis, data.jumlah || 1, data.inventaris || '', data.keluhan, data.keterangan || '', '', 'Menunggu', '', '', fileUrl, fileName, '', '' // FotoURL, FotoName (diisi admin) ]); return { success: true, id: id, fileUrl: fileUrl }; } catch (err) { return { success: false, message: err.toString() }; } } function updateRecord(data) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] == data.id) { var rowNum = i + 1; sheet.getRange(rowNum, 11).setValue(data.teknisi || ''); sheet.getRange(rowNum, 12).setValue(data.status || 'Menunggu'); sheet.getRange(rowNum, 13).setValue(data.catatan || ''); sheet.getRange(rowNum, 14).setValue(data.tglSelesai || ''); // Upload foto hasil perbaikan jika ada if (data.fotoBase64 && data.fotoName) { var folder = DriveApp.getFolderById(FOLDER_ID); var blob = Utilities.newBlob( Utilities.base64Decode(data.fotoBase64), data.fotoMime || 'image/jpeg', data.id + '_foto_' + data.fotoName ); var file = folder.createFile(blob); file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); sheet.getRange(rowNum, 17).setValue(file.getUrl()); sheet.getRange(rowNum, 18).setValue(data.fotoName); } return { success: true }; } } return { success: false, message: 'Tiket tidak ditemukan' }; } catch (err) { return { success: false, message: err.toString() }; } } function deleteRecord(id) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); var rows = sheet.getDataRange().getValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] == id) { sheet.deleteRow(i + 1); return { success: true }; } } return { success: false, message: 'Tiket tidak ditemukan' }; } catch (err) { return { success: false, message: err.toString() }; } } function generateId() { var year = new Date().getFullYear(); var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(SHEET_NAME); var data = sheet.getDataRange().getValues(); var count = 0; for (var i = 1; i < data.length; i++) { if (String(data[i][0]).indexOf('TIK-' + year) === 0) count++; } return 'TIK-' + year + String(count + 1).padStart(4, '0'); }
